# 세션 3: SET / GET / DEL / EXISTS
# difftest 는 줄마다 redis-cli 를 새로 띄웁니다. 즉 줄마다 새 커넥션입니다.
# 앞 줄에서 SET 한 값이 다음 줄 GET 에 보이면, 커넥션끼리 저장소를 같이 쓴다는 뜻입니다.
#
# 주의: redis-cli 출력을 비교하므로 nil($-1)과 빈 문자열($0), 정수(:1)와 문자열 "1"을 구분하지 못합니다.
# 바이트 단위 검사는 src/command.rs 의 단위 테스트에 있습니다.

SET name hello
GET name

# 덮어쓰기. 커맨드 이름은 대소문자를 가리지 않습니다
set name world
GET name

# 없는 키는 Null Bulk String
GET nope

# 키는 대소문자를 가립니다
GET NAME

# 바이트 그대로 저장합니다
SET 한글 값
GET 한글

# EXISTS 는 있는 키의 개수. 같은 키를 여러 번 쓰면 그만큼 셉니다
EXISTS name
EXISTS nope
EXISTS name name nope

# DEL 은 실제로 지운 개수
DEL name
GET name
DEL name
SET a 1
SET b 2
DEL a b c
EXISTS a b

# 같은 키를 두 번 써도 한 번만 지워집니다 (EXISTS 와 다릅니다)
SET k v
DEL k k

# 인자 개수
SET
SET k
GET
GET k extra
DEL
EXISTS

# SET 옵션은 S04 입니다. 모르는 옵션은 syntax error 입니다
SET k v foo

# S01, S02 가 깨지지 않았는지
PING
ECHO hello
